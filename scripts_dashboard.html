Encerrar Tarefa Travada (SDK)

Finaliza uma tarefa que ficou em loop ou travada no Maestro usando o SDK Python nativo.

from botcity.maestro import BotMaestroSDK, AutomationTaskFinishStatus

# Inicializa o SDK usando as variaveis do arquivo .env
maestro = BotMaestroSDK.from_sys_args()

# Se estiver testando localmente sem argumentos de linha de comando:
# maestro = BotMaestroSDK(
#     server="https://developers.botcity.dev", 
#     login="seu-login-maestro", 
#     key="sua-chave-maestro"
# )
# maestro.login()

task_id = "[TASK_ID]"

print(f"Finalizando task {task_id} no Maestro...")
maestro.finish_task(
    task_id=task_id,
    status=AutomationTaskFinishStatus.[STATUS],
    message="[MESSAGE]"
)
print("Executado com sucesso!")

Consultar Status da Tarefa (REST API)

Faz uma chamada HTTP direta para a API do Orchestrator para ler os metadados atuais de uma tarefa.

import os
import requests

# Tenta carregar credenciais do .env
server = os.getenv("BOTCITY_SERVER", "developers.botcity.dev")
if not server.startswith("http"):
    server = f"https://{server}"

login = os.getenv("BOTCITY_LOGIN")
key = os.getenv("BOTCITY_KEY")
workspace = os.getenv("BOTCITY_WORKSPACE", login)

# 1. Login para adquirir JWT Token
login_url = f"{server}/api/v2/workspace/{workspace}/login"
print(f"Efetuando login em {login_url}...")
response = requests.post(login_url, json={"login": login, "key": key})
response.raise_for_status()
token = response.json()["accessToken"]

# 2. Consultar dados da Tarefa
task_id = "[TASK_ID]"
task_url = f"{server}/api/v2/task/{task_id}"
headers = {
    "Authorization": f"Bearer {token}",
    "organization": workspace
}

print(f"Requisitando dados da task {task_id}...")
task_resp = requests.get(task_url, headers=headers)
task_resp.raise_for_status()

task_data = task_resp.json()
print("\n--- Informacoes da Tarefa ---")
print(f"ID: {task_data.get('id')}")
print(f"Status: {task_data.get('status')}")
print(f"Atividade: {task_data.get('activityId')}")
print(f"Mensagem: {task_data.get('finishMessage')}")
print(f"Data de Criacao: {task_data.get('creationDate')}")

Inserir Item no Datapool (SDK)

Adiciona programaticamente um novo item a uma fila de processamento (datapool) existente no Maestro.

from botcity.maestro import BotMaestroSDK

# Inicializa o SDK usando as variaveis do arquivo .env
maestro = BotMaestroSDK.from_sys_args()

datapool_label = "[DATAPOOL_LABEL]"
item_values = [DATAPOOL_ITEM]

print(f"Inserindo item no Datapool '{datapool_label}'...")
item = maestro.create_datapool_item(
    datapool_label=datapool_label,
    values=item_values
)

if item:
    print(f"Item inserido com sucesso! ID no Maestro: {item.id}")
else:
    print("Falha ao criar o item no Datapool.")

Disparar Alerta no Painel (SDK)

Envia uma notificação visual (Info, Warn, Error) vinculada a uma execução de tarefa específica.

from botcity.maestro import BotMaestroSDK, AlertType

# Inicializa o SDK usando as variaveis do arquivo .env
maestro = BotMaestroSDK.from_sys_args()

task_id = "[TASK_ID]"

print("Enviando alerta ao Maestro...")
maestro.alert(
    task_id=task_id,
    title="[ALERT_TITLE]",
    message="[MESSAGE]",
    alert_type=AlertType.[ALERT_TYPE]
)
print("Alerta registrado!")

Fazer Upload de Artefato (SDK)

Faz upload de relatórios, prints ou planilhas geradas pelo bot diretamente para a aba "Result Files" da tarefa.

from botcity.maestro import BotMaestroSDK

# Inicializa o SDK usando as variaveis do arquivo .env
maestro = BotMaestroSDK.from_sys_args()

task_id = "[TASK_ID]"
nome_no_maestro = "relatorio_execucao.xlsx"
caminho_do_arquivo = "./output/relatorio_execucao.xlsx"

print(f"Fazendo upload de '{caminho_do_arquivo}' como '{nome_no_maestro}' para a task {task_id}...")
try:
    maestro.post_artifact(
        task_id=task_id,
        artifact_name=nome_no_maestro,
        filepath=caminho_do_arquivo
    )
    print("Upload realizado com sucesso!")
except Exception as e:
    print(f"Erro ao realizar upload: {e}")
